35 float calcSampleAfterAM(float originalIndex, float phase)
36 {
37 int size = getAmpLfos().size();
38 float index = originalIndex * (size - 1);
39 int lowerIndex = static_cast<int>(index);
40 int upperIndex = std::min(lowerIndex + 1, size - 1);
41 float factor = index - lowerIndex;
42 float s = getAmpLfos()[lowerIndex]->processSample(phase) +
43 (getAmpLfos()[upperIndex]->processSample(phase) -
44 getAmpLfos()[lowerIndex]->processSample(phase)) *
45 factor;
46 s = juce::jlimit(-1.0f, 1.0f, s);
47 return s;

88 // 波形生成逻辑
89 auto makeTriangle = [](float x)
90 {
91 float normX = x / juce::MathConstants<float>::twoPi;
92 return 2.0f * std::abs(2.0f * (normX - std::floor(normX + 0.5f))) - 1.0f;
93 };
94
95 auto makeSaw = [](float x)
96 {
97 return juce::jmap(x, 0.0f, juce::MathConstants<float>::twoPi, -1.0f, 1.0f);
98 };
99
100 auto makeSquare = [](float x)
101 {
102 return (x < juce::MathConstants<float>::pi) ? 1.0f : -1.0f;
103 };
104
105 auto makeSine = [](float x)
106 {
107 return std::sin(x);
108 };
109
110 auto makeComplexWave = [](float x)
111 {
112 return std::sin(x);
113 };
114
115 auto makeRoundedTriangle = [](float x)
116 {
117 return 2.0f * std::abs(std::sin(juce::MathConstants<float>::pi * x)) - 1.0f;
118 };
119
120 auto makeSoftSquare = [](float x)
121 {
122 return std::tanh(3.0f * std::sin(2.0f * juce::MathConstants<float>::pi * x));
123 };
124
125 auto makePwm = [](float x)
126 {
127 return (x < 0.25f) ? 1.0f : -1.0f;
128 };

130 auto makeSmoothRand = [](float x)
131 {
132 // 平滑过渡的随机波形，适合 LFO morphing（伪 Perlin 可替代）
133 static float prev = 0.0f;
134 static float next = juce::Random::getSystemRandom().nextFloat() * 2.0f - 1.0f;
135 float frac = x - std::floor(x);
136 if (frac < 0.01f)
137 {
138 prev = next;
139 next = juce::Random::getSystemRandom().nextFloat() * 2.0f - 1.0f;
140 }
141 return juce::jmap(frac, 0.0f, 1.0f, prev, next); // 线性插值
142 };
143
144 auto makeNoise = [](float x)
145 {
146 return juce::Random::getSystemRandom().nextFloat() * 2.0f - 1.0f;
147 };
148
149 auto makeStepRand = [](float x)
150 {
151 constexpr int N = 128; // N 是步数，决定了波形的分辨率
152 int index = std::floor(x * N);
153 return juce::Random::getSystemRandom().nextFloat() * 2.0f - 1.0f; // 随机值范围 -1 到 1
154 };
