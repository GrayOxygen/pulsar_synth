99 void reloadSynthPreset(juce::AudioProcessorValueTreeState& apvts)
100 {
101 //refresh and load impulse file,
102 int impulseSwitchIndex = static_cast<int>(*apvts.getRawParameterValue(why::ParameterID::impulseSwitch));
103
104 //show the lastest template impulse option，and save convolution resource
105 int index = static_cast<int>(*apvts.getRawParameterValue(why::ParameterID::impulseTemplateFile));
106
107 if (impulseSwitchIndex == static_cast<int>(why::ImpulseSwitchEnum::Template))
108 {
109 getConvolutionResource()->saveTemplateImpulse(juce::String(index + 1));
110 }
111
112 //Display the latest sample impulse option and (must) directly load the file into the convolution resource
113 //(because switch the impulse menu option to no longer save the file)
114 String sampleImpulsePath = apvts.state.getProperty(why::PropertyID::sampleImpulsePath).toString();
115 if (sampleImpulsePath.isNotEmpty())
116 {
117 juce::File file(sampleImpulsePath);
118 if (file.existsAsFile())
119 {
120 // 文件存在，则保存资源文件到convolution resource中（全局的）
121 getConvolutionResource()->saveLastSampleFileAsBlock(file);
122 if (impulseSwitchIndex == static_cast<int>(why::ImpulseSwitchEnum::Sample))
123 {
124 //加载impulse response
125 getConvolutionResource()->loadSampleImpulseFile();
126 }
127 }
128 }
129
130 //update current play mode
131 int currentPlayModeIndex = static_cast<int>(*apvts.getRawParameterValue(why::ParameterID::playMode));
132 setCurrentPlayModeEnum(currentPlayModeIndex);
133
134 //update to the latest state of synth: like mapping the values of all parameters and properties to synth, etc
135 executeCurSynthCallback([&](std::shared_ptr<PulsarSynth>& synth)
136 {
137 synth->reloadPreset(apvts);
138 });
139 }

164 void initConvolution(double sampleRate, int samplesPerBlock, int numChannels)
165 {
166 //防止prepareToPlay多次触发更新，保证幂等性
167 if (!convolutionResource)
168 {
169 this->convolutionResource = std::make_shared<ConvolutionResource>(sampleRate, samplesPerBlock, numChannels);
170 }
171
172 for (std::shared_ptr<PulsarSynth> tempSynth : pulsarSynths)
173 {
174 for (int i = 0; i < tempSynth->getNumVoices(); ++i)
175 {
176 juce::SynthesiserVoice* voice = tempSynth->getVoice(i);
177 PulsarSynthVoice* pulsarVoice = dynamic_cast<PulsarSynthVoice*>(voice);
178 pulsarVoice->getCommonVoiceSate()->convolutionResource = convolutionResource;
179 }
180 }
181 }

187 void init(juce::AudioProcessorValueTreeState& apvts)
188 {
189 //Prevent prepareToPlay from triggering updates multiple times to achieve idempotence
190 if (!commonVoiceSate)
191 {
192 this->commonVoiceSate = std::make_shared<CommonVoiceSate>();
193 }
194
195 pulsarSynthForMidi->addSound(new PulsarSynthSound());
196 //only support 4 voices
197 for (int i = 0; i < 4; i++)
198 {
199 pulsarSynthForMidi->addVoice(new PulsarSynthVoice());
200
201 PulsarSynthVoice* v = dynamic_cast<PulsarSynthVoice*>(pulsarSynthForMidi->getVoice(i));
202 v->setCommonVoiceSate(commonVoiceSate);
203 v->connectParameters(apvts);
204 }
205
206 pulsarSynthForAuto->addSound(new PulsarSynthSound());
207 pulsarSynthForAuto->addVoice(new PulsarSynthVoice());
208
209 PulsarSynthVoice* v = dynamic_cast<PulsarSynthVoice*>(pulsarSynthForAuto->getVoice(0));
210 v->setCommonVoiceSate(commonVoiceSate);
211 v->connectParameters(apvts);
212
213 pulsarSynths.push_back(pulsarSynthForMidi);
214 pulsarSynths.push_back(pulsarSynthForAuto);
215 }
